cmake_minimum_required(VERSION 3.21)

# ROCm root: Windows = HIP SDK (%HIP_PATH%), Linux = -DROCM_PATH / $ROCM_PATH / /opt/rocm.
if(WIN32)
    if(DEFINED ENV{HIP_PATH})
        file(TO_CMAKE_PATH "$ENV{HIP_PATH}" _rocm_default)
    endif()
else()
    if(DEFINED ROCM_PATH)
        set(_rocm_default "${ROCM_PATH}")
    elseif(DEFINED ENV{ROCM_PATH})
        set(_rocm_default "$ENV{ROCM_PATH}")
    else()
        set(_rocm_default "/opt/rocm")
    endif()
endif()
set(ROCM_ROOT "${_rocm_default}" CACHE PATH "ROCm / HIP SDK install root")
if(NOT ROCM_ROOT)
    message(FATAL_ERROR "Set HIP_PATH or pass -DROCM_ROOT=<HIP SDK root>")
endif()

# GPU arch: use -DCMAKE_HIP_ARCHITECTURES=gfxXXXX, else ask the first tool that works.
if(NOT CMAKE_HIP_ARCHITECTURES)
    find_program(_agent NAMES amdgpu-arch rocm_agent_enumerator hipInfo
                 HINTS "${ROCM_ROOT}/bin" "${ROCM_ROOT}/llvm/bin" NO_DEFAULT_PATH)
    if(_agent)
        execute_process(COMMAND "${_agent}" OUTPUT_VARIABLE _out ERROR_QUIET)
        string(REGEX MATCHALL "gfx[0-9a-f]+" _archs "${_out}")  # also parses "gcnArchName: gfx1102:xnack-"
        list(REMOVE_ITEM _archs gfx000)
        list(REMOVE_DUPLICATES _archs)
    endif()
    if(NOT _archs)
        message(FATAL_ERROR "Could not detect the GPU architecture. Pass -DCMAKE_HIP_ARCHITECTURES=gfxXXXX (e.g. gfx1102)")
    endif()
    set(CMAKE_HIP_ARCHITECTURES "${_archs}" CACHE STRING "HIP GPU architectures" FORCE)
endif()
message(STATUS "HIP architectures: ${CMAKE_HIP_ARCHITECTURES}")

if(WIN32)
    # ROCm clang for C/CXX/HIP (MSVC headers/STL are still used)
    set(CMAKE_C_COMPILER   "${ROCM_ROOT}/bin/clang.exe"   CACHE FILEPATH "" FORCE)
    set(CMAKE_CXX_COMPILER "${ROCM_ROOT}/bin/clang++.exe" CACHE FILEPATH "" FORCE)
    set(CMAKE_RC_COMPILER  "${ROCM_ROOT}/bin/llvm-rc.exe" CACHE FILEPATH "" FORCE)
    set(CMAKE_HIP_COMPILER "${ROCM_ROOT}/bin/clang++.exe" CACHE FILEPATH "" FORCE)
    # MSVC STL vs clang HIP constexpr clash
    set(CMAKE_HIP_FLAGS "-D_MSVC_STL_USE_ABORT_AS_DOOM_FUNCTION -Xclang -fno-cuda-host-device-constexpr"
        CACHE STRING "HIP flags" FORCE)
else()
    if(NOT CMAKE_HIP_COMPILER AND NOT DEFINED ENV{HIPCXX} AND EXISTS "${ROCM_ROOT}/llvm/bin/clang++")
        set(CMAKE_HIP_COMPILER "${ROCM_ROOT}/llvm/bin/clang++" CACHE FILEPATH "HIP compiler")
    endif()
endif()
set(CMAKE_HIP_COMPILER_ROCM_ROOT "${ROCM_ROOT}" CACHE PATH "")
set(HIP_PLATFORM "amd" CACHE STRING "")
set(hip_DIR "${ROCM_ROOT}/lib/cmake/hip" CACHE PATH "")
list(APPEND CMAKE_PREFIX_PATH "${ROCM_ROOT}" "${ROCM_ROOT}/lib/cmake")

project(HipVulkanInteropSync LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_language(HIP)

find_package(hip CONFIG REQUIRED)
find_package(Vulkan REQUIRED)   # Windows: VULKAN_SDK env var; Linux: dev package or SDK

if(WIN32)
    add_executable(repro_d3d12_fence src/repro_d3d12_fence.hip)
    add_executable(repro_opaque_win32 src/repro_opaque.hip)
    foreach(_t repro_d3d12_fence repro_opaque_win32)
        target_link_libraries(${_t} PRIVATE hip::host Vulkan::Vulkan d3d12 dxgi)
    endforeach()
else()
    add_executable(repro_opaque_fd src/repro_opaque.hip)
    target_link_libraries(repro_opaque_fd PRIVATE hip::host Vulkan::Vulkan)
endif()
